"""mcs - Minecraft editing studio CLI.  Run `uv run mcs <command> -h` (from studio/) for details.

Assembly line:
  mcs new <slug> --game rec.mkv [--mic mic.wav]   create project
  mcs transcribe <slug>                           word-level transcript (source time)
  mcs analyze <slug>                              loudness/scene signals + frame sheets
  mcs scaffold <slug>                             first-pass edit.json (dead air removed)
  (planner edits edit.json: zooms, SFX, memes, greenscreens, text, bleeps, music)
  mcs check <slug>                                validate + timeline digest + policy warnings
  mcs render <slug> [--preview]                   ffmpeg render -> out/<slug>.mp4
Library:
  mcs assets find|show|stats                      query the editing-pack database
  mcs frames <slug> --from S --to E               frame sheet for a moment
  mcs ui                                          local web UI at http://127.0.0.1:8765
One click:
  mcs make <slug> --game rec.mkv [--mic m.wav] [--facecam cam.mp4] --title "..." --context "..."
  mcs auto <slug> [--final] [--shorts 3]          transcribe -> analyze -> autoplan -> render -> shorts
  mcs autoplan <slug> [--mode funny]              heuristic creative pass (no LLM needed)
  mcs engage <slug>                               add attention zooms to an existing edit
  mcs shorts <slug> [--count 3] [--render]        cut 9:16 shorts from the edit
Libraries:
  mcs library add|scan|status                     your Stock Library (B-roll, stock, memes)
  mcs motion list|render <template> --var k=v     channel motion-graphics templates (HyperFrames)
  mcs avatar build|preview                        narrator avatar from brand/avatar/skin.png
"""

def cmd_assets(a):
    from . import assets as A
    if a.action == "stats":
        print(json.dumps(A.stats(), indent=2))
        return
    if a.action == "show":
        for i in a.ids:
            print(json.dumps(A.get(i), indent=2, ensure_ascii=False))
        return
    q = A.Query(moments=a.moment or [], roles=a.role or [], categories=a.category or [], kinds=a.kind or [],
                key_modes=a.key_mode or [], placement=a.placement or [], text=a.text, max_rating=a.max_rating,
                max_risk=a.max_risk, min_energy=a.min_energy, max_energy=a.max_energy,
                max_duration=a.max_duration, min_duration=a.min_duration, exclude=a.exclude or [],
                project=a.project, limit=a.limit, seed=a.seed)
    res = A.find(q)
    if a.json:
        print(json.dumps(res, indent=1, ensure_ascii=False))
    else:
        for r in res:
            print(A.brief(r))
        if not res:
            print("(no matches - loosen filters, try --text, or raise --max-rating/--max-risk)")


def cmd_check(a):
    from . import assets as A
    from .edl import build_spans, load, resolve_time, total_duration
    from .paths import STYLE, project_dir
    proj = project_dir(a.slug)
    edl = load(proj / "edit.json")
    style = json.loads(STYLE.read_text(encoding="utf-8"))
    pace = style["pacing"]
    warnings, errors = [], []

    def lookup(i):
        try:
            return A.get(i)
        except KeyError:
            errors.append(f"unknown asset {i}")
            return None
    spans = build_spans(edl, lookup)
    total = total_duration(edl, lookup)
    rating_rank = {"clean": 0, "mild": 1, "explicit": 2, "banned": 3}
    max_default = rating_rank[edl.get("max_rating", pace["default_max_rating"])]
    first_max = rating_rank[pace["first_15s_max_rating"]]
    timeline = []

    def check_asset(aid, at, what):
        x = lookup(aid)
        if not x:
            return
        r = rating_rank.get(x.get("content_rating") or "explicit", 2)
        if r >= 3:
            errors.append(f"{what}: {aid} is BANNED content")
        elif at is not None and at < 15 and r > first_max:
            warnings.append(f"{what}: {aid} ({x['content_rating']}) in first 15s hurts monetization")
        elif r > max_default:
            warnings.append(f"{what}: {aid} rated {x['content_rating']} above the video's max_rating")
        if x.get("copyright_risk") == "high":
            warnings.append(f"{what}: {aid} has HIGH copyright risk (Content ID)")
        timeline.append((at if at is not None else -1, what, aid))

    t = 0.0
    for i, s in enumerate(edl["segments"]):
        if s.get("asset"):
            check_asset(s["asset"], t, f"segment {s.get('id', i)} cutaway")
        t = spans[i].out_end + s.get("freeze_end", 0.0)
    groups = [("overlay", edl.get("overlays", [])), ("sfx", edl.get("audio", {}).get("sfx", [])),
              ("music", edl.get("audio", {}).get("music", []))]
    counts = {}
    for name, evs in groups:
        counts[name] = len(evs)
        for k, ev in enumerate(evs):
            at = resolve_time(ev, spans, warnings, f"{name}[{k}]") if ("at" in ev or "src_at" in ev or "seg" in ev) else 0.0
            if ev.get("asset"):
                check_asset(ev["asset"], at, f"{name}[{k}]")
    for k, ev in enumerate(edl.get("text", [])):
        at = resolve_time(ev, spans, warnings, f"text[{k}]")
        timeline.append((at if at is not None else -1, f"text[{k}]", ev.get("text")))
    mins = max(total / 60, 1e-6)
    zooms = sum(len([z for z in s.get("zoom", []) if not z.get("auto")]) for s in edl["segments"])
    auto_zooms = sum(len([z for z in s.get("zoom", []) if z.get("auto")]) for s in edl["segments"])
    cutaways = sum(1 for s in edl["segments"] if s.get("type") == "cutaway")
    density = {
        "sfx_per_minute": (counts["sfx"] / mins, pace["sfx_per_minute"]),
        "zooms_per_minute": (zooms / mins, pace["zooms_per_minute"]),
        "meme_cutaways_per_minute": (cutaways / mins, pace["meme_cutaways_per_minute"]),
        "greenscreens_per_minute": (sum(1 for o in edl.get("overlays", []) if o["type"] == "greenscreen") / mins,
                                    pace["greenscreens_per_minute"]),
    }
    src_segs = [s for s in edl["segments"] if s.get("type", "source") == "source"]
    avg_seg = sum((s["out"] - s["in"]) / s.get("speed", 1) for s in src_segs) / max(len(src_segs), 1)
    print(f"duration {total:.1f}s ({total / 60:.1f} min), {len(edl['segments'])} segments "
          f"(avg source shot {avg_seg:.1f}s), {cutaways} cutaways, {auto_zooms / mins:.1f} engagement zooms/min")
    for k, (v, (lo, hi)) in density.items():
        flag = "" if lo <= v <= hi else ("  <- LOW" if v < lo else "  <- HIGH")
        print(f"  {k:<26} {v:5.1f}  (target {lo}-{hi}){flag}")
    if a.timeline:
        for at, what, aid in sorted(timeline, key=lambda x: x[0]):
            m, s = divmod(max(at, 0), 60)
            print(f"  {int(m):02d}:{s:05.2f}  {what:<22} {aid}")
    for w in warnings:
        print("WARN ", w)
    for e in errors:
        print("ERROR", e)
    sys.exit(1 if errors else 0)

# ------------------------------------------------------------------ helpers
def inside(base: Path, rel: str) -> Path:
    p = (base / rel).resolve()
    if base.resolve() not in p.parents and p != base.resolve():
        raise HTTPException(403, "path escapes base")
    if not p.exists():
        raise HTTPException(404, f"not found: {rel}")
    return p


def read_json(p: Path, default=None):
    try:
        return json.loads(p.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return default


def db() -> sqlite3.Connection:
    con = sqlite3.connect(ASSETS_DB)
    con.row_factory = sqlite3.Row
    return con

@app.get("/media/abs")
def media_abs(path: str):
    """Raw recordings referenced by absolute path in project.json (only those)."""
    allowed = set()
    for pj in PROJECTS.glob("*/project.json"):
        for s in (read_json(pj, {}).get("sources") or {}).values():
            if isinstance(s, dict) and s.get("file"):
                f = Path(s["file"])
                allowed.add(str((f if f.is_absolute() else pj.parent / f).resolve()))
    p = str(Path(path).resolve())
    if p not in allowed:
        raise HTTPException(403, "not a project source")
    return FileResponse(p)

@app.get("/api/stats")
def stats():
    s = A.stats()

def row_out(r: sqlite3.Row) -> dict:
    d = A.as_dict(r)
    d.pop("abs_path", None)
    return d


@app.get("/api/assets")
def list_assets(q: str = "", moment: list[str] = Query(default=[]), category: list[str] = Query(default=[]),
                role: list[str] = Query(default=[]), rating: list[str] = Query(default=[]),
                risk: list[str] = Query(default=[]), kind: str = "", key_mode: str = "", root: str = "",
                ranked: bool = False, project: str = "", limit: int = 60, offset: int = 0):
    if ranked and (moment or q or role):
        # same scorer the planner uses
        res = A.find(A.Query(moments=moment, roles=role, categories=category, text=q or None,
                             max_rating="explicit", max_risk="high", project=project or None, limit=limit))
        for r in res:
            r.pop("abs_path", None)
        return {"total": len(res), "items": res}
    where, args = ["1=1"], []
    if q:
        terms = [t for t in "".join(c if c.isalnum() else " " for c in q).split() if t]
        if terms:
            ids = [r[0] for r in db().execute("SELECT id FROM assets_fts WHERE assets_fts MATCH ?",
                                              (" ".join(f'"{t}"*' for t in terms),))]
            where.append(f"id IN ({','.join('?' * len(ids))})" if ids else "0")
            args += ids
    for col, vals in (("category", category), ("content_rating", rating), ("copyright_risk", risk)):
        if vals:
            where.append(f"{col} IN ({','.join('?' * len(vals))})")
            args += vals
    for m in moment:
        where.append("moments LIKE ?")
        args.append(f'%"{m}"%')
    for ro in role:
        where.append("roles LIKE ?")
        args.append(f'%"{ro}"%')
    if kind:
        where.append("kind = ?")
        args.append(kind)
    if root:
        where.append("root = ?")
        args.append(root)
    if key_mode:
        where.append("key_mode = ?")
        args.append(key_mode)
    con = db()
    w = " AND ".join(where)
    total = con.execute(f"SELECT COUNT(*) FROM assets WHERE {w}", args).fetchone()[0]
    rows = con.execute(f"SELECT * FROM assets WHERE {w} ORDER BY category, title LIMIT ? OFFSET ?",
                       args + [limit, offset]).fetchall()
    return {"total": total, "items": [row_out(r) for r in rows]}


@app.get("/api/assets/{aid}")
def get_asset(aid: str):
    try:
        d = A.get(aid)
    except KeyError:
        raise HTTPException(404) from None
    d.pop("abs_path", None)
    d["used_in"] = [dict(r) for r in db().execute(
        "SELECT project, COUNT(*) n FROM usage WHERE asset_id=? GROUP BY project", (aid,))]
    return d


class AssetPatch(BaseModel):
    fields: dict


EDITABLE = {"description", "moments", "roles", "tags", "quote", "mood", "energy", "content_rating",
            "copyright_risk", "placement", "key_mode", "use_window", "volume_db", "frequency", "notes"}


@app.patch("/api/assets/{aid}")
def patch_asset(aid: str, body: AssetPatch):
    bad = set(body.fields) - EDITABLE
    if bad:
        raise HTTPException(400, f"not editable: {sorted(bad)}")
    path = CATALOG / "overrides.json"
    ov = read_json(path, {})
    ov.setdefault(aid, {}).update(body.fields)
    path.write_text(json.dumps(ov, indent=2, ensure_ascii=False), encoding="utf-8")
    r = subprocess.run([sys.executable, str(CATALOG / "merge_catalog.py")], capture_output=True, text=True)
    if r.returncode != 0:
        raise HTTPException(500, r.stderr[-2000:])
    return get_asset(aid)


# ------------------------------------------------------------------ projects
def project_status(p: Path) -> dict:
    w = p / "work"
    outs = sorted((p / "out").glob("*.mp4"), key=lambda f: f.stat().st_mtime, reverse=True) if (p / "out").exists() else []
    edl = read_json(p / "edit.json")
    pj = read_json(p / "project.json", {})
    return {
        "slug": p.name, "title": pj.get("title", p.name), "duration_s": pj.get("duration_s"),
        "transcribed": (w / "transcript.json").exists(), "analyzed": (w / "signals.json").exists(),
        "has_edit": edl is not None, "segments": len(edl["segments"]) if edl else 0,
        "outputs": [{"name": f.name, "size": f.stat().st_size,
                     "mtime": datetime.fromtimestamp(f.stat().st_mtime).isoformat(timespec="seconds")} for f in outs],
        "modified": datetime.fromtimestamp(max((f.stat().st_mtime for f in p.glob("*.json")), default=p.stat().st_mtime)).isoformat(timespec="seconds"),
        "job": next((j.public() for j in reversed(JOBS.values()) if j.slug == p.name and j.status == "running"), None),
        "kind": pj.get("kind", "episode"), "parent": pj.get("parent"), "context": pj.get("context", ""),
        "generator": (edl or {}).get("generator"), "format": (edl or {}).get("format", "landscape"),
        "shorts": sorted(c.name for c in PROJECTS.glob(f"{p.name}--short-*") if (c / "edit.json").exists()),
    }


@app.get("/api/projects")
def projects():
    out = [project_status(p) for p in PROJECTS.iterdir() if (p / "project.json").exists()]
    return sorted(out, key=lambda x: x["modified"], reverse=True)

def new_project(body: NewProject):
    import re
    if not re.fullmatch(r"[a-z0-9][a-z0-9_-]{1,40}", body.slug):
        raise HTTPException(400, "slug: lowercase letters, digits, - and _ only")
    from .ingest import new_project as np_
    try:
        np_(body.slug, body.game, body.mic or None, body.mic_stream, 0, body.mic_offset, body.title, False,
            facecam=body.facecam or None, facecam_offset=body.facecam_offset, context=body.context)
    except SystemExit as e:
        raise HTTPException(400, str(e)) from None
    return project_status(PROJECTS / body.slug)

@app.get("/api/projects/{slug}/timeline")
def timeline(slug: str):
    """EDL resolved into output time, grouped by track, for the timeline view."""
    p = inside(PROJECTS, slug)
    edl = read_json(p / "edit.json")
    if not edl:
        return {"duration": 0, "tracks": {}, "warnings": []}
    warnings: list[str] = []

    def lookup(i):
        try:
            return A.get(i)
        except KeyError:
            warnings.append(f"unknown asset {i}")
            return None
    try:
        validate(edl)
        spans = build_spans(edl, lookup)
    except (EDLError, KeyError, TypeError) as e:
        return {"duration": 0, "tracks": {}, "warnings": [f"EDL error: {e}"]}
    tracks: dict[str, list] = {"segments": [], "overlays": [], "text": [], "sfx": [], "music": [], "bleeps": [],
                               "zooms": []}
    t = 0.0
    for i, s in enumerate(edl["segments"]):
        d = seg_duration(s, lookup)
        tracks["segments"].append({"i": i, "start": t, "end": t + d, "type": s.get("type", "source"),
                                   "label": s.get("id") or s.get("asset") or f"seg{i}", "src_in": s.get("in"),
                                   "src_out": s.get("out"), "speed": s.get("speed", 1), "effects": s.get("effects", []),
                                   "mode": s.get("mode") or edl.get("mode", "funny")})
        for z in s.get("zoom", []) + [dict(x, shake=True) for x in s.get("shake", [])]:
            a = z["at"] if "at" in z else (z["src_at"] - s["in"]) / s.get("speed", 1)
            dz = z.get("dur") or (z.get("in", 0.12) + z.get("hold", 0.6) + z.get("out", 0.18))
            tracks["zooms"].append({"start": t + a, "end": t + min(a + dz, d),
                                    "label": "shake" if z.get("shake") else f"x{z.get('to', 1.3)}"})
        t += d
    total = t

    def add(track, evs, default_dur, label_fn):
        for k, ev in enumerate(evs):
            at = resolve_time(ev, spans, warnings, f"{track}[{k}]") if ("at" in ev or "src_at" in ev) else 0.0
            if at is None:
                continue
            a = lookup(ev["asset"]) if ev.get("asset") else None
            dur = ev.get("dur") or (ev.get("until", total) - at if track == "music" else None) \
                or (a.get("tail_s") or a.get("duration_s") if a else None) or default_dur
            tracks[track].append({"k": k, "start": at, "end": min(total, at + dur), "label": label_fn(ev, a),
                                  "asset": ev.get("asset"), "thumb": a.get("thumb") if a else None})
    add("overlays", edl.get("overlays", []), 1.5, lambda e, a: f"{e['type']}: {(a or {}).get('title') or e.get('file')}")
    add("text", edl.get("text", []), 1.2, lambda e, a: e.get("text", ""))
    add("sfx", edl.get("audio", {}).get("sfx", []), 1.0, lambda e, a: (a or {}).get("title") or e.get("asset"))
    add("music", edl.get("audio", {}).get("music", []), total, lambda e, a: (a or {}).get("title") or e.get("asset"))
    add("bleeps", edl.get("audio", {}).get("bleeps", []), 0.4, lambda e, a: "bleep")
    return {"duration": total, "tracks": tracks, "warnings": warnings, "format": edl.get("format", "landscape")}


class EditBody(BaseModel):
    edit: dict


@app.put("/api/projects/{slug}/edit")
def save_edit(slug: str, body: EditBody):
    p = inside(PROJECTS, slug)
    try:
        validate(body.edit)
    except (EDLError, KeyError, TypeError) as e:
        raise HTTPException(400, f"invalid EDL: {e}") from None
    cur = p / "edit.json"
    if cur.exists():
        hist = p / "history"
        hist.mkdir(exist_ok=True)
        shutil.copy2(cur, hist / f"edit.{datetime.now():%Y%m%d-%H%M%S}.json")
    cur.write_text(json.dumps(body.edit, indent=2, ensure_ascii=False), encoding="utf-8")
    return {"ok": True}


# ------------------------------------------------------------------ jobs
class Job:
    def __init__(self, slug: str, stage: str, cmd: list[str], cwd: Path):
        self.id = uuid.uuid4().hex[:8]
        self.slug, self.stage, self.cmd, self.cwd = slug, stage, cmd, cwd
        self.status = "running"
        self.lines: list[str] = []
        self.started = time.time()
        self.ended: float | None = None
        self.rc: int | None = None
        self.proc: subprocess.Popen | None = None
        threading.Thread(target=self._run, daemon=True).start()

    def _run(self):
        env = dict(os.environ, PYTHONUNBUFFERED="1", PYTHONIOENCODING="utf-8")
        try:
            self.proc = subprocess.Popen(self.cmd, cwd=self.cwd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                                         text=True, encoding="utf-8", errors="replace", env=env)
            for line in self.proc.stdout:
                self.lines.append(line.rstrip())
                if len(self.lines) > 4000:
                    del self.lines[:1000]
            self.rc = self.proc.wait()
            self.status = "done" if self.rc == 0 else "failed"
        except Exception as e:  # noqa: BLE001
            self.lines.append(f"!! {e}")
            self.status = "failed"
        self.ended = time.time()

    def public(self, tail: int = 200) -> dict:
        return {"id": self.id, "slug": self.slug, "stage": self.stage, "status": self.status, "rc": self.rc,
                "started": self.started, "elapsed": round((self.ended or time.time()) - self.started, 1),
                "log": self.lines[-tail:]}

class RunBody(BaseModel):
    prompt: str | None = None


@app.post("/api/projects/{slug}/run/{stage}")
def run_stage(slug: str, stage: str, body: RunBody | None = None):
    inside(PROJECTS, slug)
    if any(j.slug == slug and j.status == "running" for j in JOBS.values()):
        raise HTTPException(409, "a job is already running for this project")
    if stage == "plan":
        cb = claude_bin()
        if not cb:
            raise HTTPException(400, "claude CLI not on PATH - copy the prompt into Claude Code instead")
        prompt = (body.prompt if body and body.prompt else plan_prompt(slug))
        cmd = [cb, "-p", prompt, "--allowedTools",
               "Read,Glob,Grep,Edit,Write,Bash(uv run mcs:*),Bash(ffmpeg:*),Bash(ffprobe:*)"]
        job = Job(slug, stage, cmd, STUDIO.parent)
    elif stage in STAGES:
        job = Job(slug, stage, [sys.executable, "-m", "mcstudio.cli", *STAGES[stage][:1], slug, *STAGES[stage][1:]], STUDIO)
    else:
        raise HTTPException(400, f"unknown stage {stage}")
    JOBS[job.id] = job
    return job.public()

        raise HTTPException(404)

@app.post("/api/jobs/{jid}/cancel")
def cancel(jid: str):
    j = JOBS.get(jid)
    if j and j.proc and j.status == "running":
        subprocess.run(["taskkill", "/F", "/T", "/PID", str(j.proc.pid)], capture_output=True)
        j.status = "cancelled"
    return {"ok": True}

@app.post("/api/library/upload")
async def library_upload(folder: str = Form("Inbox"), files: list[UploadFile] = File(...)):
    if folder not in LIB_FOLDERS:
        raise HTTPException(400, f"folder must be one of {LIB_FOLDERS}")
    dest = ROOT / "Stock Library" / folder
    dest.mkdir(parents=True, exist_ok=True)
    saved = []
    for f in files:
        target = dest / Path(f.filename or "upload.bin").name
        with target.open("wb") as out:
            while chunk := await f.read(1 << 20):
                out.write(chunk)
        saved.append(target.name)
    job = Job("_library", "library-scan", [sys.executable, "-m", "mcstudio.cli", "library", "scan"], STUDIO)
    JOBS[job.id] = job
    return {"saved": saved, "job": job.public()}

@app.get("/api/fs")
def fs(path: str = ""):
    p = Path(path).expanduser() if path else Path.home() / "Videos"
    if not p.exists():
        p = Path.home()
    p = p.resolve()
    entries = []
    try:
        for c in sorted(p.iterdir(), key=lambda c: (not c.is_dir(), c.name.lower())):
            if c.name.startswith("."):
                continue
            if c.is_dir() or c.suffix.lower() in {".mkv", ".mp4", ".mov", ".wav", ".mp3", ".flac", ".m4a", ".webm"}:
                st = c.stat()
                entries.append({"name": c.name, "path": str(c), "dir": c.is_dir(), "size": st.st_size,
                                "mtime": datetime.fromtimestamp(st.st_mtime).isoformat(timespec="minutes")})
    except PermissionError:
        pass
    return {"path": str(p), "parent": str(p.parent) if p.parent != p else None, "entries": entries}

def serve(host: str = "127.0.0.1", port: int = 8765, open_browser: bool = True):
    import uvicorn
    if open_browser:
        threading.Timer(1.2, lambda: __import__("webbrowser").open(f"http://{host}:{port}")).start()
    uvicorn.run(app, host=host, port=port, log_level="warning")

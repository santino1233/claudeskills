        if (LIB.favorites && !a.favorite) load();

async function toggleFavorite(a, on) {   // `on` comes from the DOM, so card + drawer never disagree
  try {
    const r = await api(`/api/assets/${encodeURIComponent(a.id)}/favorite`, { method: "POST", body: { on } });
    a.favorite = r.favorite;
    markFavorite(r.id, r.favorite);
    toast(r.favorite ? "★ Favourited: the auto-editor will prefer it" : "Removed from favourites");
    return r;
  } catch (e) { toast(e.message, true); }
}
function markFavorite(id, on) {   // sync every card + the drawer button showing this asset
  $$(`.asset[data-id="${CSS.escape(id)}"]`).forEach((el) => {
    el.classList.toggle("fav", on);
    const s = $(".star", el); s.textContent = on ? "★" : "☆"; s.title = on ? "Unfavourite" : "Favourite";
  });
  const b = $(`#favBtn[data-id="${CSS.escape(id)}"]`);
  if (b) { b.classList.toggle("on", on); b.textContent = on ? "★ Favourited" : "☆ Favourite"; }
}

function playAsset(a, btn) {
  const url = packUrl(a.path, a.root);
  if (!player.paused && player.dataset.id === a.id) { player.pause(); btn.textContent = "▶"; return; }
  $$(".play").forEach((b) => (b.textContent = "▶"));
  player.src = url; player.dataset.id = a.id;
  player.currentTime = 0; player.play().catch(() => toast("Can't play this format in the browser", true));
  btn.textContent = "❚❚";
  player.onended = () => (btn.textContent = "▶");
}

    $("#regen").disabled = true;
    const tick = async () => {

function motionField(v, val) {
  const id = `mv-${v.key}`, lbl = esc(v.label || v.key), key = esc(v.key);
  if (v.options?.length) {
    const opts = v.options.map((o) => (typeof o === "object" ? o : { value: o, label: o }));
    return `<label class="field">${lbl}<select id="${id}" data-key="${key}" data-type="${esc(v.type)}">${opts.map((o) => `<option value="${esc(o.value)}" ${String(o.value) === String(val) ? "selected" : ""}>${esc(o.label ?? o.value)}</option>`).join("")}</select></label>`;
  }
  if (v.type === "boolean") return `<label class="row check"><input type="checkbox" id="${id}" data-key="${key}" data-type="boolean" ${val ? "checked" : ""}> ${lbl}</label>`;
  if (v.type === "color") {
    const hex = /^#[0-9a-f]{6}$/i.test(val || "") ? val : "#ffffff";
    return `<label class="field">${lbl}<div class="row" style="gap:6px"><input type="color" id="${id}-pick" value="${esc(hex.toLowerCase())}">
      <input type="text" id="${id}" data-key="${key}" data-type="color" value="${esc(val ?? "")}" style="width:110px" class="mono">
      ${v.brand ? `<span class="faint" style="font-size:11px">brand ${esc(v.brand.replace("colors.", ""))}</span>` : ""}</div></label>`;
  }
  if (v.type === "number") return `<label class="field">${lbl}<input type="number" id="${id}" data-key="${key}" data-type="number" value="${esc(val ?? "")}" step="${v.step ?? "any"}" ${v.min != null ? `min="${v.min}"` : ""} ${v.max != null ? `max="${v.max}"` : ""}></label>`;
  return `<label class="field">${lbl}<input type="text" id="${id}" data-key="${key}" data-type="string" value="${esc(val ?? "")}"></label>`;
}

"" : "disabled title=\"not available\""}>${o === "vertical" ? "▮ Vertical 9:16" : "▬ Landscape 16:9"}</button>`).join("")}</div></div>
      <div class="row"><label class="row check" title="Half resolution, 30fps: quick to check. Untick for the 1080p60 file you'll use."><input type="checkbox" id="mpreview" ${saved._preview === false ? "" : "checked"}> Quick preview (540p30)</label>

  d.hidden = false;
  $("#dclose").onclick = closeDrawer;

  const values = () => {
    const out = {};
    $$("#mform [data-key]").forEach((el) => {
      const k = el.dataset.key, ty = el.dataset.type;
      out[k] = ty === "boolean" ? el.checked : ty === "number" ? (el.value === "" ? null : +el.value) : el.value;
    });
    return out;
  };
  const changed = () => {   // only vars that differ from the template defaults go into the snippet
    const all = values(), o = {};
    t.vars.forEach((v) => { if (all[v.key] !== null && String(all[v.key]) !== String(v.default ?? "")) o[v.key] = all[v.key]; });
    return o;
  };
  const snippet = () => {
    const s = { type: "motion", template: t.name, vars: changed(), seg: "<segment id>", rel: 0.0 };
    const dur = $("#mdur")?.value; if (dur) s.dur = +dur;
    return JSON.stringify(s);
  };
  const remember = () => {
    MOTION_FORM[t.name] = { ...values(), _orientation: orientation, _preview: $("#mpreview").checked, _duration: $("#mdur")?.value || "" };
    store.set("motionForm", MOTION_FORM);
    $("#medl").value = snippet();
  };
  $$("#mform input, #mform select").forEach((el) => el.addEventListener("input", remember));
  $$("#mform input[type=color]").forEach((pk) => {
    const txt = $("#" + CSS.escape(pk.id.replace(/-pick$/, "")));
    pk.addEventListener("input", () => { txt.value = pk.value.toUpperCase(); remember(); });
    txt.addEventListener("input", () => { if (/^#[0-9a-f]{6}$/i.test(txt.value)) pk.value = txt.value.toLowerCase(); });
  });
  $$("#morient button").forEach((b) => (b.onclick = () => { orientation = b.dataset.o; $$("#morient button").forEach((x) => x.classList.toggle("on", x === b)); remember(); }));
  $("#mreset").onclick = () => { delete MOTION_FORM[t.name]; store.set("motionForm", MOTION_FORM); openMotion(t); };
  $("#mcopy").onclick = () => navigator.clipboard.writeText($("#medl").value).then(() => toast("EDL snippet copied"), () => toast("Couldn't copy: select the text instead", true));
  $("#medl").value = snippet();

  $("#mrender").onclick = async () => {
    const vars = values();
    Object.keys(vars).forEach((k) => vars[k] === null && delete vars[k]);
    const body = { name: t.name, vars, orientation, preview: $("#mpreview").checked };
    const dur = $("#mdur")?.value; if (dur) body.duration = +dur;
    let j;
    try { j = await api("/api/motion/render", { method: "POST", body }); } catch (e) { return toast(e.message, true); }
    const btn = $("#mrender"), log = $("#mlog"), res = $("#mresult");
    btn.disabled = true; btn.textContent = "Rendering…"; log.hidden = false; res.hidden = true;
    const tick = async () => {
      if (!document.body.contains(log)) { clearInterval(iv); return; }
      const r = await api("/api/jobs/" + j.id).catch(() => null); if (!r) return;
      log.textContent = `$ render ${t.name} (${r.status}, ${r.elapsed}s)\n` + r.log.join("\n");
      log.scrollTop = log.scrollHeight;
      if (r.status === "running") return;
      clearInterval(iv); btn.disabled = false; btn.textContent = "Render";
      if (r.status !== "done" || !r.result) return toast(`Render ${r.status}`, true);
      const o = r.result;
      $("#mprev").innerHTML = `<video class="${o.orientation === "vertical" ? "mvert" : ""}" src="${o.mp4_url}" autoplay muted loop playsinline controls></video>`;
      res.hidden = false;
      res.innerHTML = `<div class="row"><span class="badge r-clean">rendered</span><span class="faint" style="font-size:12px">${o.orientation} · ${o.preview ? "preview 540p30" : "full 1080p60"} · ${fmtSize(o.size)} · shown over gameplay</span></div>
        <div class="row"><a class="btn gold" href="${o.mov_url}" download="${esc(o.mov.split("/").pop())}">⬇ Download .mov (alpha)</a><button class="btn small" id="mcopyPath">Copy path</button></div>
        <code class="mono faint" style="word-break:break-all">${esc(o.mov)}</code>`;
      $("#mcopyPath").onclick = () => { navigator.clipboard.writeText(o.mov); toast("Path copied"); };
      toast("Rendered " + t.name);
    };
    const iv = setInterval(tick, 1000); pollers.push(iv); tick();
  };
}
